<?php
$script = '/usr/local/emhttp/plugins/unraid-security/scripts/security-check.sh';
$output = is_executable($script) ? shell_exec($script . ' 2>/dev/null') : null;
$data = $output ? json_decode($output, true) : null;
$findings = is_array($data) && isset($data['findings']) && is_array($data['findings']) ? $data['findings'] : [];

function us_e($value) {
    return htmlspecialchars((string)$value, ENT_QUOTES | ENT_SUBSTITUTE, 'UTF-8');
}

function us_count($findings, $status) {
    $count = 0;
    foreach ($findings as $finding) {
        if (($finding['status'] ?? '') === $status) {
            $count++;
        }
    }
    return $count;
}

$failCount = us_count($findings, 'fail');
$warningCount = us_count($findings, 'warning');
$unknownCount = us_count($findings, 'unknown');
$passCount = us_count($findings, 'pass');
$overall = $failCount > 0 ? 'Needs attention' : ($warningCount > 0 ? 'Review recommended' : ($unknownCount > 0 ? 'Review incomplete' : 'No major findings'));
?>
